## 3. PROJECT INITIALIZATION & TAILWIND CSS SETUP

Run the following commands from the repository root:

```bash
# 1. Create frontend directory with Angular 18 standalone template
npx -y @angular/cli@18 new frontend --style=css --routing=false --standalone=true --skip-tests=true --ssr=false

# 2. Navigate into frontend
cd frontend

# 3. Install Lucide Icons for high-end UI graphics
npm install lucide-angular clsx

# 4. Install Tailwind CSS and Autoprefixer
npm install -D tailwindcss postcss autoprefixer
npx tailwindcss init -p
```

```typescript
import { Injectable, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable } from 'rxjs';
import { StockCheckRequest, StockCheckResponse } from '../models/stock.model';
import { environment } from '../../environments/environment';

@Injectable({
  providedIn: 'root'
})
export class StockService {
  private http = inject(HttpClient);
  private apiUrl = `${environment.apiUrl}/api/stock`;

  checkStock(request: StockCheckRequest): Observable<StockCheckResponse> {
    return this.http.post<StockCheckResponse>(`${this.apiUrl}/check`, request);
  }
}
```

```typescript
import { Component, inject, signal } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { StockService } from '../../services/stock.service';
import { ComplianceStandard, ComplianceStatus, StockCheckResponse } from '../../models/stock.model';

@Component({
  selector: 'app-stock-screener',
  standalone: true,
  imports: [CommonModule, FormsModule],
  templateUrl: './stock-screener.component.html'
})
export class StockScreenerComponent {
  private stockService = inject(StockService);

  ticker = signal<string>('AAPL');
  selectedStandard = signal<ComplianceStandard>(ComplianceStandard.Aaoifi);
  isLoading = signal<boolean>(false);
  errorMessage = signal<string | null>(null);
  stockResult = signal<StockCheckResponse | null>(null);

  ComplianceStandard = ComplianceStandard;
  ComplianceStatus = ComplianceStatus;

  quickTickers = ['AAPL', 'TSLA', 'MSFT', '2222.SR', 'AMZN'];

  onSelectQuickTicker(symbol: string) {
    this.ticker.set(symbol);
    this.runScreening();
  }

  runScreening() {
    if (!this.ticker().trim()) return;

    this.isLoading.set(true);
    this.errorMessage.set(null);

    this.stockService.checkStock({
      ticker: this.ticker().trim().toUpperCase(),
      standard: Number(this.selectedStandard())
    }).subscribe({
      next: (res) => {
        this.stockResult.set(res);
        this.isLoading.set(false);
      },
      error: (err) => {
        this.errorMessage.set(err?.error?.error || 'Failed to retrieve stock financial compliance data.');
        this.isLoading.set(false);
      }
    });
  }
}
```

### File: `frontend/src/app/components/fiqh-advisor/fiqh-advisor.component.ts`
```typescript
import { Component, inject, signal } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { FiqhService } from '../../services/fiqh.service';
import { FiqhQueryResponse } from '../../models/fiqh.model';

@Component({
  selector: 'app-fiqh-advisor',
  standalone: true,
  imports: [CommonModule, FormsModule],
  templateUrl: './fiqh-advisor.component.html'
})
export class FiqhAdvisorComponent {
  private fiqhService = inject(FiqhService);

  question = signal<string>('Can I do dropshipping where I sell goods to a customer before buying or possessing them?');
  isLoading = signal<boolean>(false);
  errorMessage = signal<string | null>(null);
  fiqhResponse = signal<FiqhQueryResponse | null>(null);

  sampleQuestions = [
    'Can I do dropshipping where I sell goods before taking ownership?',
    'Is it permissible to charge late payment penalty fees on commercial invoices?',
    'What are the Shariah requirements for earning affiliate marketing commissions?',
    'Can partners agree to a fixed guaranteed return in a Musharakah?'
  ];

  selectSampleQuestion(q: string) {
    this.question.set(q);
    this.askAdvisor();
  }

  askAdvisor() {
    if (!this.question().trim()) return;

    this.isLoading.set(true);
    this.errorMessage.set(null);

    this.fiqhService.askFiqhQuestion({ question: this.question().trim() }).subscribe({
      next: (res) => {
        this.fiqhResponse.set(res);
        this.isLoading.set(false);
      },
      error: (err) => {
        this.errorMessage.set(err?.error?.error || 'Failed to query RAG fiqh knowledge bases.');
        this.isLoading.set(false);
      }
    });
  }
}
```

## 10. PRODUCTION BUILD & RENDER DEPLOYMENT GUIDE

### Option A: Unified Monorepo (Single Render Web Service - Recommended)
When using the root `Dockerfile` provided in `backend.txt`:
1. The multi-stage Docker build automatically compiles Angular (`dist/tijarah-frontend/browser`) and copies it into `.NET`'s `wwwroot`.
2. Both frontend and backend run on a single Render free Web Service instance with zero CORS issues and zero extra cost.

### Option B: Render Free Static Site
If you want to host Angular separately as a Render Static Site:
1. In Render Dashboard, click **New +** -> **Static Site**.
2. Connect your GitHub repository.
3. Root Directory: `frontend`.
4. Build Command: `npm install && npm run build -- --configuration production`.
5. Publish Directory: `dist/tijarah-frontend/browser`.
6. Update `frontend/src/environments/environment.prod.ts` with your backend Render URL:
   ```typescript
   export const environment = {
     production: true,
     apiUrl: 'https://tijarah-api.onrender.com'
   };
   ```
